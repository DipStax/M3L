cmake_minimum_required(VERSION 3.24)

# 2D Render
add_executable(Graphics2D
    2d_render.cpp
)

target_compile_options(Graphics2D INTERFACE /W4)
target_compile_definitions(Graphics2D INTERFACE NOMINMAX)
target_link_libraries(Graphics2D M3L_GRAPHICS)
target_include_directories(Graphics2D INTERFACE ${M3L_INCLUDE_DIRS})
