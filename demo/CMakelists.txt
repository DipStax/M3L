cmake_minimum_required(VERSION 3.24)

# 2D Render
add_executable(demo_2drender
    2d_render.cpp
)

target_compile_options(demo_2drender INTERFACE /W4)
target_compile_definitions(demo_2drender INTERFACE NOMINMAX)
target_link_libraries(demo_2drender M3L)
target_include_directories(demo_2drender INTERFACE ${M3L_INCLUDE_DIRS})
